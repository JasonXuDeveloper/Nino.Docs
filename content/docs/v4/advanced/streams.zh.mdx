Nino 序列化到 `IBufferWriter<byte>`，从 `ReadOnlySpan<byte>` 反序列化。流、异步 I/O、多段输入、序列化回调和压缩，在这两个调用之上多写几行就能实现。本页为每一种都给出了可以直接复制的写法。

## 流 [#streams]

流没有消息边界，所以每个值都写成一帧：先是 4 字节小端序的长度，再是数据。`Writer` 先预留长度的位置，接着序列化值，最后回填长度，全部在同一个缓冲区里完成：

```csharp title="StreamFrames.cs"
using System;
using System.Buffers;
using System.Buffers.Binary;
using System.IO;
using System.Threading;
using System.Threading.Tasks;
using Nino.Core;

public static partial class StreamFrames
{
    public const int MaxFrameSize = 16 << 20; // 一帧（也就是一个对端）最多能让读取方租用的内存

    // [int32 长度][负载]，在 buffer 中构建
    public static void Frame<T>(T value, NinoArrayBufferWriter buffer)
    {
        buffer.ResetWrittenCount();
        var writer = new Writer(buffer);
        int slot = writer.ReserveInt32();
        NinoSerializer.Serialize(value, ref writer); // [!code highlight]
        writer.PatchLength(slot);   // 槽位之后写入的字节数
        writer.Flush();
        writer.ReleaseReferences(); // 结束这次调用，和 NinoSerializer 一样
    }

    public static void Write<T>(Stream stream, T value, NinoArrayBufferWriter buffer)
    {
        Frame(value, buffer);
        stream.Write(buffer.WrittenSpan);
    }
}
```

读取正好反过来：先读 4 字节前缀，检查长度，再读入恰好这么多字节，然后反序列化。流一次返回的字节可能比请求的少，所以 `Fill` 会循环读取，直到读满为止：

```csharp title="StreamFrames.cs"
using System;
using System.Buffers;
using System.Buffers.Binary;
using System.IO;
using Nino.Core;

public static partial class StreamFrames
{
    /// <summary>把下一帧读入 value（默认新建一个）。到达流末尾时返回 false。</summary>
    public static bool TryRead<T>(Stream stream, ref T value)
    {
        byte[] buffer = ArrayPool<byte>.Shared.Rent(4096); // 先放前缀，再放负载
        try
        {
            if (!Fill(stream, buffer, 4, endAllowed: true))
                return false;                              // 在两帧之间正常结束
            int length = ReadLength(buffer);
            if (length > buffer.Length)
                Grow(ref buffer, length);
            Fill(stream, buffer, length, endAllowed: false);
            NinoDeserializer.Deserialize(buffer.AsSpan(0, length), ref value); // [!code highlight]
            return true;
        }
        finally
        {
            ArrayPool<byte>.Shared.Return(buffer);
        }
    }

    public static T Read<T>(Stream stream)
    {
        T value = default;
        if (!TryRead(stream, ref value))
            throw new EndOfStreamException();
        return value;
    }

    // .NET 7+ 上 Stream.ReadAtLeast 做的是同样的事；这个循环在任何地方都能运行，包括 Unity
    private static bool Fill(Stream stream, byte[] buffer, int count, bool endAllowed)
    {
        for (int read = 0; read < count; )
        {
            int n = stream.Read(buffer, read, count - read);
            if (n == 0)
            {
                if (read == 0 && endAllowed)
                    return false;
                throw new EndOfStreamException();
            }
            read += n;
        }
        return true;
    }

    private static void Grow(ref byte[] buffer, int length)
    {
        byte[] larger = ArrayPool<byte>.Shared.Rent(length);
        ArrayPool<byte>.Shared.Return(buffer);
        buffer = larger;
    }

    private static int ReadLength(ReadOnlySpan<byte> prefix)
    {
        int length = BinaryPrimitives.ReadInt32LittleEndian(prefix);
        if ((uint)length > MaxFrameSize)
            throw new InvalidDataException($"Frame of {length} bytes");
        return length;
    }
}
```

```csharp
using var buffer = new NinoArrayBufferWriter(); // 每个线程或连接一个
using (var file = File.Create(path))
    StreamFrames.Write(file, checkpoint, buffer);

using (var file = File.OpenRead(path))
    checkpoint = StreamFrames.Read<Checkpoint>(file);
```

`TryRead` 每帧租一个池化数组来读前缀和数据，并直接读入 `value`：等 `value` 的结构对上之后，写一帧、读一帧，除了内容变了的字符串，什么都不分配。`Read` 每次返回新值。

一串值（其他库用 `IAsyncEnumerable<T>` 流式传输的那种）就是连续的帧：每个元素写一帧，读到流结束为止，每一帧都读进同一个实例：

```csharp
AuditEvent auditEvent = null;                   // 每一帧都原地填充它
while (StreamFrames.TryRead(stream, ref auditEvent))
    Index(auditEvent);
```

帧格式与[网络层指南](/zh/docs/v4/advanced/networking#frame-the-stream)相同，两边可以混用。

## 异步 [#async]

`Writer` 是 `ref struct`，不能用在 `async` 方法里，所以先用 `Frame` 同步构建帧，再 await I/O。`WriteAsync` 本身不需要 `async`：它直接返回流的任务，所以在任务完成之前不要复用缓冲区。把下面几个方法加到 `StreamFrames` 中：

```csharp title="StreamFrames.cs"
using System;
using System.Buffers;
using System.IO;
using System.Threading;
using System.Threading.Tasks;
using Nino.Core;

public static partial class StreamFrames
{
    /// <summary>返回的任务完成之前，不要复用 buffer。</summary>
    public static ValueTask WriteAsync<T>(Stream stream, T value, NinoArrayBufferWriter buffer,
        CancellationToken ct = default)
    {
        Frame(value, buffer);
        return stream.WriteAsync(buffer.WrittenMemory, ct);
    }

    public static async ValueTask<T> ReadAsync<T>(Stream stream, CancellationToken ct = default)
    {
        byte[] buffer = ArrayPool<byte>.Shared.Rent(4096);
        try
        {
            await FillAsync(stream, buffer, 4, ct);
            int length = ReadLength(buffer);
            if (length > buffer.Length)
                Grow(ref buffer, length);
            await FillAsync(stream, buffer, length, ct);
            return NinoDeserializer.Deserialize<T>(buffer.AsSpan(0, length));
        }
        finally
        {
            ArrayPool<byte>.Shared.Return(buffer);
        }
    }

    private static async ValueTask FillAsync(Stream stream, byte[] buffer, int count, CancellationToken ct)
    {
        for (int read = 0; read < count; )
        {
            int n = await stream.ReadAsync(buffer.AsMemory(read, count - read), ct);
            if (n == 0) throw new EndOfStreamException();
            read += n;
        }
    }
}
```

`async` 方法只有在真正等待 I/O 时才会分配状态机；`ReadAsync` 每帧返回一个新值（`async` 方法不能带 `ref` 参数）。不允许分配内存的循环，请用同步的 `TryRead` 或 [`PipeReader`](/zh/docs/v4/advanced/networking#reading-frames)。

`Fill` 和 `FillAsync` 在所有目标上都能用；在 .NET 7 及以上，也可以换成 `Stream.ReadAtLeast` / `ReadExactlyAsync`。`stream.Write(span)`、`WriteAsync(memory)` 和 `ReadAsync(memory)` 需要 .NET Standard 2.1，Nino 支持的所有 Unity 版本（2022.3 及以上）都满足；只面向 .NET Standard 2.0 时，改用 `byte[]` 重载，写入的数组用 `MemoryMarshal.TryGetArray(buffer.WrittenMemory, out var segment)` 取出。

使用 `System.IO.Pipelines` 时，直接序列化到 `PipeWriter`（它就是 `IBufferWriter<byte>`），再从 `PipeReader` 中读取帧。完整的读取循环参见[读取帧](/zh/docs/v4/advanced/networking#reading-frames)，它解码每一帧的方式与下面的工具方法相同。

## 多段输入（`ReadOnlySequence`） [#readonlysequence-input]

`PipeReader` 或 `SequenceReader` 给出的是 `ReadOnlySequence<byte>`。只有一段时直接从它反序列化；有多段时，先拷贝一次到池化的缓冲区。`ref` 重载会填充你保留的实例：

```csharp
public static T Deserialize<T>(in ReadOnlySequence<byte> data)
{
    T value = default;
    Deserialize(data, ref value);
    return value;
}

public static void Deserialize<T>(in ReadOnlySequence<byte> data, ref T value)
{
    if (data.IsSingleSegment)
    {
        NinoDeserializer.Deserialize(data.First.Span, ref value); // .NET 5+ 上可用 data.FirstSpan
        return;
    }

    int length = checked((int)data.Length);
    byte[] rented = ArrayPool<byte>.Shared.Rent(length);
    try
    {
        data.CopyTo(rented);
        NinoDeserializer.Deserialize(rented.AsSpan(0, length), ref value);
    }
    finally
    {
        ArrayPool<byte>.Shared.Return(rented);
    }
}
```

把这两个方法也加到 `StreamFrames` 中。管道每次读取通常只返回一段，所以大多数帧走第一个分支，不需要任何拷贝。

## 回调 [#callbacks]

Nino 自己不调用任何钩子。把前后要执行的逻辑放进一个接口，再写一个小工具方法，在调用 Nino 的前后调用它们。泛型参数约束为这个接口，所以调用不会给结构体装箱：

```csharp
public interface ISerializationCallbacks
{
    void OnBeforeSerialize();
    void OnAfterDeserialize();
}

public static class CallbackSerializer
{
    public static void Serialize<T>(T value, IBufferWriter<byte> output) where T : ISerializationCallbacks
    {
        value?.OnBeforeSerialize();
        NinoSerializer.Serialize(value, output);
    }

    public static void Deserialize<T>(ReadOnlySpan<byte> data, ref T value) where T : ISerializationCallbacks
    {
        NinoDeserializer.Deserialize(data, ref value);
        value?.OnAfterDeserialize();
    }
}
```

- 泛型工具方法里的调用不走[拦截器](/zh/docs/v4/performance#interceptors)，因为那里的 `T` 是类型参数。这些调用走常规路径，每次多花几纳秒。
- 嵌套对象的钩子也由你的工具方法负责：在根对象的 `OnAfterDeserialize` 中调用它们。
- 属于构造过程的逻辑，可以放在 [`[NinoConstructor]`](/zh/docs/v4/constructors) 或属性 setter 里，它们在反序列化过程中就会执行。

## 压缩 [#compression]

压缩 Nino 写出的字节；读取前先解压到池化的缓冲区。使用 span 时，每帧开头写两个长度：压缩后的长度告诉读取方这一帧在哪里结束（同一个缓冲区里可以连着放多帧），压缩前的长度告诉它要租多大的缓冲区。`Compress` 可以写入任意 `IBufferWriter<byte>`（另一个你保留的 `NinoArrayBufferWriter`、`PipeWriter`），`Decompress` 填充你保留的实例，所以缓冲区增长到位之后两者都不分配内存：

```csharp title="BrotliPayload.cs"
using System;
using System.Buffers;
using System.Buffers.Binary;
using System.IO;
using System.IO.Compression;
using Nino.Core;

public static class BrotliPayload
{
    public const int MaxSize = 16 << 20; // 一帧最多能让读取方租用的内存

    /// <summary>写出 [int32 压缩后长度][int32 原始长度][Brotli 数据]。scratch 存放未压缩的字节。</summary>
    public static void Compress<T>(T value, NinoArrayBufferWriter scratch, IBufferWriter<byte> output)
    {
        scratch.ResetWrittenCount();
        NinoSerializer.Serialize(value, scratch);

        int length = scratch.WrittenCount;
        Span<byte> target = output.GetSpan(8 + BrotliEncoder.GetMaxCompressedLength(length));
        // 质量取 0-11 中的 1：快，依然紧凑；窗口 22 是默认值
        if (!BrotliEncoder.TryCompress(scratch.WrittenSpan, target.Slice(8), out int written, 1, 22))
            throw new InvalidOperationException("Compression failed");
        BinaryPrimitives.WriteInt32LittleEndian(target, written);
        BinaryPrimitives.WriteInt32LittleEndian(target.Slice(4), length);
        output.Advance(8 + written);
    }

    /// <summary>读出一帧到 value，返回这一帧的字节数，下一帧从那里开始。</summary>
    public static int Decompress<T>(ReadOnlySpan<byte> data, ref T value)
    {
        if (data.Length < 8)
            throw new InvalidDataException("Truncated frame");
        int compressed = BinaryPrimitives.ReadInt32LittleEndian(data);
        int length = BinaryPrimitives.ReadInt32LittleEndian(data.Slice(4));
        if ((uint)compressed > (uint)(data.Length - 8) || (uint)length > MaxSize)
            throw new InvalidDataException("Bad frame header");

        // MaxSize 是一帧最多能让读取方租用的内存：发送方不可信时把它设小
        byte[] rented = ArrayPool<byte>.Shared.Rent(length);
        try
        {
            // 一次性解压，不创建解码器实例：不分配任何对象
            if (!BrotliDecoder.TryDecompress(data.Slice(8, compressed), rented.AsSpan(0, length), out int written)
                || written != length)
                throw new InvalidDataException("Corrupt data");
            NinoDeserializer.Deserialize(rented.AsSpan(0, length), ref value);
        }
        finally
        {
            ArrayPool<byte>.Shared.Return(rented);
        }
        return 8 + compressed;
    }
}
```

使用流时，把流包装一层，再复用上面的帧工具方法。`GZipStream` 在所有平台上都可用，包括 .NET Standard 2.0 和 Unity：

```csharp
using (var gzip = new GZipStream(File.Create(path), CompressionLevel.Fastest))
    StreamFrames.Write(gzip, checkpoint, buffer);

using (var gzip = new GZipStream(File.OpenRead(path), CompressionMode.Decompress))
    checkpoint = StreamFrames.Read<Checkpoint>(gzip);
```

`BrotliEncoder`、`BrotliDecoder` 和 `BrotliStream` 需要 .NET Core 2.1 及以上，.NET Standard 中没有。如果想要更快、可以接受压缩率低一些，可以用 LZ4 库，它们提供和 `BrotliEncoder.TryCompress` 一样的 span 输入、span 输出接口。
